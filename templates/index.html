function downloadCIO() { const select = document.getElementById("historyDate"); let day = null; if (select && select.value) { day = select.value; } else if ( typeof currentState !== "undefined" && currentState && currentState.date ) { day = currentState.date; } if (!day) { alert( "No CIO date available for download." ); return; } const downloadUrl = "/api/download/cio/" + encodeURIComponent(day); const link = document.createElement("a"); link.href = downloadUrl; link.download = "Pratik_Analysis_CIO_" + day + ".xlsx"; document.body.appendChild(link); link.click(); document.body.removeChild(link); }
